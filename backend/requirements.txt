# ARQAU backend — runtime dependencies.
#
# These pins are the versions the Phase 1-3.4 suite was actually verified
# against (287/287). Do not change a pin without re-running:
#     python -m pytest tests -q
# from this directory.
#
# Dev/test-only packages live in requirements-dev.txt and are NOT installed in
# the production image.
#
# Interpreter: supported on CPython 3.12 and 3.13. Production is pinned to
# 3.12.11 via .python-version / runtime.txt / render.yaml.

# --- Web framework -------------------------------------------------------
fastapi==0.142.2
# Transitive via fastapi (which declares only "starlette>=0.46.0", no upper
# bound). Pinned so a fresh install cannot drift a major Starlette version out
# from under the tested combination.
starlette==1.7.0
# Declared without the [standard] extra on purpose: the extras (uvloop,
# httptools, watchfiles, websockets, PyYAML) are not part of the tested set.
# Adding them is a separate, deliberate production-performance change.
uvicorn==0.54.0

# --- Persistence ---------------------------------------------------------
sqlalchemy==2.1.3
alembic==1.20.0
# Default (non-production) database driver: sqlite+aiosqlite://
aiosqlite==0.22.1
# Production database driver: postgresql+asyncpg:// (required when
# APP_ENV=production). 0.31.0 ships cp312/cp313 wheels; the previously pinned
# 0.29.0 has no wheels and must be compiled from source at install time.
asyncpg==0.31.0

# --- Auth / crypto -------------------------------------------------------
# HS256 JWT signing and verification. The [cryptography] extra was removed:
# it was declared but never installed, and HS256 needs only stdlib hmac.
python-jose==3.5.0
passlib[bcrypt]==1.7.4
# Pinned to 4.0.1 deliberately. passlib 1.7.4 reads bcrypt.__about__ to detect
# the backend version (passlib/handlers/bcrypt.py:620); bcrypt >= 4.1 removed
# that attribute, so bcrypt 4.3.0 makes passlib log a spurious
# "(trapped) error reading bcrypt version" warning on every CryptContext
# creation. Hashing and verification behaviour is identical on both versions.
# 4.0.1 ships an abi3 (cp36-abi3) wheel, so it installs on 3.12 and 3.13
# without compiling. Do not "upgrade" this pin without re-checking passlib.
bcrypt==4.0.1

# --- Validation / settings ----------------------------------------------
pydantic==2.13.5
# Required by pydantic EmailStr (app/schemas/auth.py).
email-validator==2.3.0
python-dotenv==1.2.4

# --- AI provider (Phase 3.3 / 3.4) --------------------------------------
# Pinned to the version the 287-test baseline was verified against.
# The Phase 3.3/3.4 code uses only the API surface that is identical in
# openai 1.40.0 and 3.x: AsyncOpenAI(api_key=, base_url=, timeout=),
# chat.completions.create(..., response_format=, max_tokens=, temperature=,
# timeout=), the ResponseFormatJSONSchema union, and the APIConnectionError /
# APIStatusError / APITimeoutError / RateLimitError constructors.
# Note: openai >= 2.x depends on httpx2, not httpx. httpx is a runtime
# dependency here (AI Copilot `app/services/ai_copilot.py` uses httpx for
# provider payload fetching); it was previously declared in
# requirements-dev.txt.
openai==3.24.0
greenlet==3.5.6
httpx==0.28.1
